function useReveal() {
  useEffect(() => {
    const els = document.querySelectorAll(".reveal");
    const io = new IntersectionObserver((es) => es.forEach((e) => e.isIntersecting && (e.target.classList.add("in"), io.unobserve(e.target))), { threshold: 0.12 });
    els.forEach((el) => io.observe(el));
    return () => io.disconnect();
  }, []);
}

                  </div>
                </Field>
                <Field label="แนะนำตัวสั้น ๆ (ไม่บังคับ)"><textarea className="input-dark min-h-24 py-3" maxLength={1000} value={f.intro} onChange={(e) => set("intro", e.target.value)} placeholder="เล่าเกี่ยวกับตัวคุณ หรือเหตุผลที่สนใจ" /></Field>
              </Card>

              <Card title="ความสามารถพิเศษ (ทำอาหาร, ดนตรี ฯลฯ)" err={errs.talents}>
                <Field label="เลือกได้หลายข้อ (อย่างน้อย 1)">
                  <Chips options={TALENTS} value={f.talents ?? []} onChange={(v) => set("talents", v)} />
                </Field>
              </Card>

              <Card title="กิจกรรมและเวลาที่สะดวกรับลูกค้า (วันไหน สัปดาห์ละกี่วัน วันละกี่ชั่วโมง)">
                <Field label="กิจกรรมที่ทำได้ (เลือกได้หลายข้อ)" err={errs.activities}>
                  <Chips options={ACTIVITIES} value={f.activities ?? []} onChange={(v) => set("activities", v)} />
                </Field>
                {f.package === "B" && (
                  <Field label="มีใบรับรองการนวด?" err={errs.massage_cert}>
                    <Seg options={["มี", "ไม่มี"]} value={f.massage_cert === "yes" ? "มี" : f.massage_cert === "no" ? "ไม่มี" : ""} onChange={(v) => set("massage_cert", v === "มี" ? "yes" : "no")} />
                  </Field>
                )}
                {f.package === "B" && f.massage_cert === "yes" && (
                  <div className="w-1/2"><Upload label="ใบรับรองการนวด" hint="ไม่บังคับ" value={f.cert_doc ?? ""} err={errs.cert_doc} onPick={(file) => onFile("cert_doc", file)} /></div>
                )}
                <Field label="เวลาที่สะดวกรับลูกค้า" err={errs.avail_slots}>
                  <Chips options={TIME_SLOTS} value={f.avail_slots ?? []} onChange={(v) => set("avail_slots", v)} />
                </Field>
                <div className="grid grid-cols-2 gap-3">
                  <Field label="สัปดาห์ละกี่วัน (1–7)" err={errs.avail_days}><input className="input-dark h-12" inputMode="numeric" maxLength={1} value={f.avail_days ?? ""} onChange={(e) => set("avail_days", e.target.value.replace(/[^1-7]/g, ""))} /></Field>
                  <Field label="กี่ชม. / วัน" err={errs.avail_hours}><input className="input-dark h-12" inputMode="numeric" maxLength={2} value={f.avail_hours ?? ""} onChange={(e) => set("avail_hours", e.target.value.replace(/\D/g, ""))} /></Field>
                </div>
              </Card>

              <Card title="รูปถ่ายและเอกสาร">
                <div className="grid grid-cols-2 gap-3">
                  <Upload label="รูปถ่ายหน้าตรง" hint="เห็นหน้าชัด" value={f.photo} err={errs.photo} onPick={(file) => onFile("photo", file)} />
                  <Upload label="บัตรประชาชน" hint="สำเนาพร้อมลงนาม" value={f.id_card} err={errs.id_card} onPick={(file) => onFile("id_card", file)} />
                </div>
                <p className="text-[12px] text-[#9a8a85] -mt-1">ข้อมูลอื่น เช่น บัญชีรับเงินและที่อยู่ ทีมงานจะขอเพิ่มตอนอบรม</p>
              </Card>

              <Card title="เงื่อนไขและความยินยอม">
                <ul className="space-y-2 text-[13px] text-ink-300 leading-relaxed">
                  {TERMS_SUMMARY.map((t) => <li key={t} className="flex gap-2.5"><span className="w-1.5 h-1.5 rounded-full bg-gold-500 mt-[8px] shrink-0" /><span>{t}</span></li>)}
                </ul>
                <Consent checked={f.accept_terms} onChange={(v) => set("accept_terms", v)} required>
                  ข้าพเจ้าได้อ่านและยอมรับ <a href="/join/terms" target="_blank" rel="noopener" className="text-gold-300 underline underline-offset-2">ข้อกำหนดและเงื่อนไขการสมัคร</a> รวมถึงนโยบายการคืนเงิน
                </Consent>
                <Consent checked={f.consent_pdpa} onChange={(v) => set("consent_pdpa", v)} required>
                  ข้าพเจ้ายินยอมให้เก็บ ใช้ และเปิดเผยข้อมูลส่วนบุคคล รวมถึงข้อมูลอ่อนไหวบนสำเนาบัตรประชาชน เพื่อพิจารณาการสมัครและยืนยันตัวตน ตาม <a href="/join/privacy" target="_blank" rel="noopener" className="text-gold-300 underline underline-offset-2">นโยบายความเป็นส่วนตัว</a>
                </Consent>
                <Consent checked={f.consent_criminal} onChange={(v) => set("consent_criminal", v)} required>
                  ข้าพเจ้ายินยอมให้บริษัทตรวจสอบประวัติอาชญากรรม เพื่อความปลอดภัยของลูกค้าและผู้ให้บริการ
                </Consent>
                {errs.consent_criminal && <p data-err="1" className="text-[12px] text-crimson-300 -mt-1">{errs.consent_criminal}</p>}
                <Consent checked={!!f.ack_hours} onChange={(v) => set("ack_hours", v)} required>
                  รับทราบว่าการให้บริการอยู่ที่ {f.package ? PACKAGES[f.package].hours : "3–8 (A) / 3–5 (B)"} ชม. แล้วแต่แผนโปรโมท
                </Consent>
                <Consent checked={!!f.ack_no_refuse} onChange={(v) => set("ack_no_refuse", v)} required>
                  {ACK_NO_REFUSE_TEXT}
                </Consent>
                {(errs.ack_hours || errs.ack_no_refuse) && <p data-err="1" className="text-[12px] text-crimson-300 -mt-1">ต้องติ๊กรับทราบเงื่อนไขทั้งสองข้อ</p>}
                <Consent checked={f.consent_marketing} onChange={(v) => set("consent_marketing", v)}>
                  ยินยอมรับข่าวสาร โปรโมชัน และรอบรับสมัครถัดไป (ไม่บังคับ ถอนได้ทุกเมื่อ)
                </Consent>
              </Card>

              <button onClick={next} disabled={!f.accept_terms || !f.consent_pdpa || !f.consent_criminal || !f.ack_hours || !f.ack_no_refuse || closed} className="btn-crimson h-14 font-display font-semibold text-[17px]">
                {closed ? "ปิดรับสมัครแล้ว" : f.accept_terms && f.consent_pdpa && f.consent_criminal && f.ack_hours && f.ack_no_refuse ? "ถัดไป · ชำระค่าสมัคร →" : "ติ๊กยอมรับ 5 ข้อที่จำเป็นก่อน"}
              </button>
            </div>
          </section>
        </div>

        <footer className="px-6 pt-2 pb-8 text-center">
          <nav className="flex justify-center items-center gap-1 text-[13px]">
            <a href="/join/privacy" className="min-h-10 px-2 inline-flex items-center text-gold-300">นโยบายความเป็นส่วนตัว</a>
            <span className="text-ink-500" aria-hidden>·</span>
            <a href="/join/terms" className="min-h-10 px-2 inline-flex items-center text-gold-300">ข้อกำหนดการสมัคร</a>
          </nav>
          <p className="text-[11.5px] text-[#9a8a85] leading-relaxed mt-2">
            ข้อมูลของคุณได้รับการคุ้มครองตาม<br />พ.ร.บ.คุ้มครองข้อมูลส่วนบุคคล พ.ศ. 2562
          </p>
          <p className="text-[11px] text-ink-500 mt-3">© {new Date().getFullYear()} {BRAND.name}</p>
        </footer>

        {/* ── แถบจองลอยด้านล่าง ── */}
        <div className={`fixed bottom-0 left-1/2 -translate-x-1/2 w-full max-w-[480px] z-30 px-4 pb-[max(14px,env(safe-area-inset-bottom))] pt-3 transition-all duration-300 ${showBar && !closed ? "translate-y-0 opacity-100" : "translate-y-full opacity-0 pointer-events-none"}`}>
          <div className="flex items-center gap-3 rounded-full bg-ink-800/95 backdrop-blur-xl border border-gold-500/30 pl-5 pr-1.5 py-1.5 shadow-[0_12px_32px_rgba(0,0,0,.5)]">
            <div className="shrink-0 max-w-[38%] min-w-0">
              <p className="text-[11px] text-ink-300 truncate">{sel ? sel.name : `เหลือ ${left} ที่นั่ง`}</p>
              <p className="font-display text-ink-100 text-lg font-bold leading-tight">{sel ? `฿${baht(sel.price)}` : `เริ่ม ฿${baht(PACKAGES.A.price)}`}</p>
            </div>
            <button onClick={toForm} className="btn-crimson min-h-12 py-2 px-4 font-semibold text-[13px] leading-snug flex-1">{CTA_LABEL}</button>
          </div>
        </div>
      </main>
    </div>
  );
}

function SectionHead({ kicker, title, sub }: { kicker: string; title: string; sub?: string }) {
  return (
    <div className="reveal">
      <p className="text-[11px] font-semibold tracking-[0.25em] text-gold-400">{kicker}</p>
      <h2 className="font-display text-[26px] font-semibold mt-1 leading-tight">{title}</h2>
      {sub && <p className="text-[13px] text-ink-300 mt-1">{sub}</p>}
    </div>
  );
}

function Steps({ at }: { at: 1 | 2 | 3 }) {
  return (
    <div className="flex gap-1.5 mt-4" aria-label={`ขั้นตอน ${at} จาก 3`}>
      {["ใบสมัคร", "ชำระเงิน", "รับรหัส MB"].map((s, i) => (
        <div key={s} className="flex-1">
          <div className={`h-1.5 rounded-full ${i < at ? "bg-gradient-to-r from-gold-400 to-gold-600" : "bg-ink-700"}`} />
          <p className={`text-[11px] mt-1.5 ${i + 1 === at ? "text-gold-300" : "text-[#9a8a85]"}`}>{s}</p>
        </div>
      ))}
    </div>
  );
}

function Seg({ options, value, onChange }: { options: readonly string[]; value: string; onChange: (v: string) => void }) {
  return (
    <div className="grid gap-2" style={{ gridTemplateColumns: `repeat(${options.length}, minmax(0, 1fr))` }}>
      {options.map((x) => (
        <button key={x} type="button" aria-pressed={value === x} onClick={() => onChange(x)}
          className={`h-11 rounded-full border text-[13px] transition ${value === x ? "border-gold-500 bg-gold-500 text-[#1a1206] font-semibold" : "border-white/15"}`}>{x}</button>
      ))}
    </div>
  );
}

function Upload({ label, hint, value, err, onPick }: { label: string; hint: string; value: string; err?: string; onPick: (f?: File) => void }) {
  return (
    <label className="flex flex-col gap-1.5 cursor-pointer min-w-0" data-err={err ? "1" : undefined}>
      <span className={`relative aspect-[4/5] rounded-2xl overflow-hidden grid place-items-center border-2 bg-white/[0.03] ${value ? "border-gold-500" : err ? "border-crimson-500 border-dashed" : "border-gold-500/40 border-dashed"}`}>
        {value
          // eslint-disable-next-line @next/next/no-img-element
          ? <img src={value} alt={label} className="absolute inset-0 w-full h-full object-cover" />
          : <span className="flex flex-col items-center gap-1.5 text-gold-400">
              <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.7" className="w-8 h-8"><path d="M4 8h3l2-3h6l2 3h3v11H4z" /><circle cx="12" cy="13" r="3.5" /></svg>
              <span className="text-[11px]">แตะเพื่อเลือก</span>
            </span>}
        {value && <span className="absolute bottom-1.5 right-1.5 w-6 h-6 rounded-full bg-gold-500 text-[#1a1206] grid place-items-center text-xs font-bold">✓</span>}
      </span>
      <span className="text-[13px] font-medium leading-tight">{label}</span>
      <span className="text-[11px] text-[#9a8a85] leading-tight -mt-1">{hint}</span>
      {err && <span role="alert" className="text-[11px] text-crimson-300">{err}</span>}
      <input type="file" accept="image/jpeg,image/png,image/webp" className="sr-only" onChange={(e) => onPick(e.target.files?.[0])} />
    </label>
  );
}

function Consent({ checked, onChange, required, children }: { checked: boolean; onChange: (v: boolean) => void; required?: boolean; children: React.ReactNode }) {
  return (
    <label className={`flex items-start gap-3 rounded-2xl border px-4 py-3.5 cursor-pointer transition ${checked ? "border-gold-500 bg-gold-500/10" : "border-white/15"}`}>
      <input type="checkbox" checked={checked} onChange={(e) => onChange(e.target.checked)} className="w-5 h-5 mt-0.5 accent-[#d4af37] shrink-0" />
      <span className="text-[13.5px] leading-relaxed">
        {children}
        {required && <span className="text-crimson-300"> *</span>}
      </span>
    </label>
  );
}

function Card({ title, err, children }: { title: string; err?: string; children: React.ReactNode }) {
  return (
    <div className={`rounded-3xl border p-4 flex flex-col gap-4 bg-white/[0.03] ${err ? "border-crimson-500" : "border-ink-600"}`} data-err={err ? "1" : undefined}>
      <p className="text-[15px] font-semibold">{title}</p>
      {children}
      {err && <span role="alert" className="text-xs text-crimson-300 -mt-2">{err}</span>}
    </div>
  );
}

function Field({ label, err, children }: { label: string; err?: string; children: React.ReactNode }) {
  return (
    <div className="flex flex-col gap-1.5 min-w-0" data-err={err ? "1" : undefined}>
      <span className="text-[13px] text-ink-300">{label}</span>
      {children}
      {err && <span role="alert" className="text-xs text-crimson-300">{err}</span>}
    </div>
  );
}

function Chips({ options, value, onChange }: { options: readonly string[]; value: string[]; onChange: (v: string[]) => void }) {
  return (
    <div className="flex flex-wrap gap-2">
      {options.map((x) => {
        const on = value.includes(x);
        return (
          <button key={x} type="button" aria-pressed={on} onClick={() => onChange(on ? value.filter((s) => s !== x) : [...value, x])}
            className={`h-10 px-4 rounded-full border text-[13px] transition ${on ? "border-gold-500 bg-gold-500/15 text-gold-300" : "border-white/15"}`}>{on ? "✓ " : ""}{x}</button>
        );
      })}
    </div>
  );
}
